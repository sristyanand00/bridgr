# Slim dependency set for Render's free tier (512MB RAM).
#
# requirements.txt is still the full set — use it locally and anywhere with real
# memory. The difference is what a 512MB container can hold:
#
#   torch + sentence-transformers   ~400MB resident   -> dropped
#   firebase-admin (grpcio, etc.)   ~150MB installed  -> dropped
#   scikit-learn                    ~50MB             -> dropped
#
# What that costs: the semantic (embedding) tier of skill extraction is off, so
# extraction runs on the PhraseMatcher + fuzzy tiers only — roughly 0.59 F1
# instead of 0.67 per evals/. Scores stay correctly normalised; see the weight
# renormalisation in core_ml/matching.py. Auth endpoints return 503, which the
# frontend already handles.

fastapi>=0.115.0
uvicorn[standard]==0.29.0
python-multipart==0.0.9

# Resume parsing. PyMuPDF is kept deliberately: it is a compact C extension and
# without it the parser falls back to pdfplumber, which handles multi-column
# resume layouts noticeably worse.
pdfplumber==0.11.0
PyMuPDF>=1.24.4

# NLP. spaCy drives the PhraseMatcher tier that does the extraction work now.
spacy>=3.8.0

numpy>=2.0.0
pandas>=2.3.0

python-dotenv==1.0.1
pydantic-settings==2.2.1
SQLAlchemy==2.0.30
slowapi==0.1.9

# Bounded LRU cache for embedding lookups; MatchingEngine degrades to an
# unbounded dict without it.
cachetools>=5.3.0

# LLM enrichment. Both are optional at runtime and guarded by try/except in
# services/llm_service.py — groq is a thin httpx client kept as a fallback for
# when the Gemini free-tier quota runs out.
google-generativeai==0.8.3
groq==0.9.0
